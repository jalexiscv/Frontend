Row

Fila horizontal del sistema de grilla. Agrupa columnas y define el espaciado entre ellas.

Codigo PHP

PHP
use Higgs\Frontend\Frontend;
$frontend = new Frontend('bootstrap', '5.3.3');
$bootstrap = $frontend->get_Builder();

// Fila con columnas responsivas
echo $bootstrap->row([
    'content' => [
        $bootstrap->col(['content' => 'Columna A', 'md' => 4]),
        $bootstrap->col(['content' => 'Columna B', 'md' => 4]),
        $bootstrap->col(['content' => 'Columna C', 'md' => 4]),
    ],
])->render();

// Fila con columnas a diferentes breakpoints
echo $bootstrap->row([
    'content' => [
        $bootstrap->col(['content' => 'Sidebar', 'md' => 3]),
        $bootstrap->col(['content' => 'Main', 'md' => 9]),
    ],
    'class' => 'mb-3',
])->render();
