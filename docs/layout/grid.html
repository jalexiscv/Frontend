Grid

Sistema de grilla automatica para distribuir elementos en columnas uniformes con espaciado configurable.

Codigo PHP

PHP
use Higgs\Frontend\Frontend;
$frontend = new Frontend('bootstrap', '5.3.3');
$bootstrap = $frontend->get_Builder();

// Grid de 3 columnas con gap de 3
echo $bootstrap->grid([
    'columns' => 3,
    'gap'     => 3,
    'content' => [
        $bootstrap->card(['content' => 'Item 1', 'icon' => 'cube']),
        $bootstrap->card(['content' => 'Item 2', 'icon' => 'cogs']),
        $bootstrap->card(['content' => 'Item 3', 'icon' => 'chart-bar']),
    ],
])->render();

// Grid de 4 columnas responsivo
echo $bootstrap->grid([
    'columns' => 4,
    'gap'     => 2,
    'content' => [
        $bootstrap->card(['content' => 'A']),
        $bootstrap->card(['content' => 'B']),
        $bootstrap->card(['content' => 'C']),
        $bootstrap->card(['content' => 'D']),
    ],
])->render();
