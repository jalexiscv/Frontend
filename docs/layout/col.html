Col

Columna responsiva con breakpoints sm, md, lg, xl. Controla el ancho en cada tamano de pantalla.

Codigo PHP

PHP
use Higgs\Frontend\Frontend;
$frontend = new Frontend('bootstrap', '5.3.3');
$bootstrap = $frontend->get_Builder();

// Columna con breakpoints responsivos
echo $bootstrap->col([
    'content' => 'Contenido',
    'sm'      => 12,
    'md'      => 6,
    'lg'      => 4,
])->render();

// Columna con offset
echo $bootstrap->col([
    'content' => 'Centrado',
    'md'      => 6,
    'offset'  => 3,
])->render();

// Row con multiples columnas
echo $bootstrap->row([
    'content' => [
        $bootstrap->col(['content' => 'Sidebar', 'md' => 3]),
        $bootstrap->col(['content' => 'Contenido principal', 'md' => 9]),
    ],
])->render();
